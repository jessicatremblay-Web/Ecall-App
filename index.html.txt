    const STORAGE_ACTIVE = 'ecall_active';
    const STORAGE_ARCHIVE = 'ecall_archive';

    if (!localStorage.getItem(STORAGE_ACTIVE)) {
      localStorage.setItem(STORAGE_ACTIVE, JSON.stringify([]));
    }
    if (!localStorage.getItem(STORAGE_ARCHIVE)) {
      localStorage.setItem(STORAGE_ARCHIVE, JSON.stringify([]));
    }

    const serverMock = {
      getActiveCallSheets: function() { return JSON.parse(localStorage.getItem(STORAGE_ACTIVE)); },
      getArchivedCallSheets: function() { return JSON.parse(localStorage.getItem(STORAGE_ARCHIVE)); },
      saveActiveCallSheets: function(data) { localStorage.setItem(STORAGE_ACTIVE, JSON.stringify(data)); },
      saveArchivedCallSheets: function(data) { localStorage.setItem(STORAGE_ARCHIVE, JSON.stringify(data)); }
    };

    let activeDataCache = [];
    let archiveDataCache = [];
    let weeklyChartInstance = null;

    function switchTab(tabName) {
      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
      const selectedTab = Array.from(document.querySelectorAll('.tab')).find(t => t.textContent.toLowerCase().includes(tabName));
      if (selectedTab) selectedTab.classList.add('active');
      document.getElementById('tab-' + tabName).classList.add('active');

      if (tabName === 'actives') loadActives();
      if (tabName === 'archives') loadArchives();
      if (tabName === 'kpis') loadKpis();
    }

    function submitForm(e) {
      e.preventDefault();
      const form = document.getElementById('callSheetForm');
      const formData = new FormData(form);
      const dataObj = Object.fromEntries(formData.entries());
      
      // Générer un ID unique
      const d = new Date();
      const randomNum = Math.floor(1000 + Math.random() * 9000);
      const newId = "CS-" + d.getFullYear().toString().substr(-2) + 
                    (d.getMonth()+1).toString().padStart(2, '0') + 
                    d.getDate().toString().padStart(2, '0') + "-" + randomNum;
      
      const newCallSheet = {
        ID: newId,
        ZAP: dataObj.zap,
        OF: dataObj.of,
        OPE: dataObj.ope,
        MSN: dataObj.msn,
        Date_Creation: dataObj.date,
        Nom_Operateur: dataObj.nomOperateur,
        Matricule: dataObj.matricule,
        FS: dataObj.fs,
        Stringer: dataObj.stringer,
        BL: dataObj.bl,
        Location: dataObj.location,
        Quoi: dataObj.quoi,
        Pourquoi: dataObj.pourquoi,
        Comment: dataObj.comment,
        Responsabilite: dataObj.responsabilite,
        SolutionProposee: dataObj.solutionProposee,
        Status: "Ouvert"
      };

      let activeData = serverMock.getActiveCallSheets();
      activeData.unshift(newCallSheet); // Ajouter au début
      serverMock.saveActiveCallSheets(activeData);
      activeDataCache = activeData;

      alert('✅ Créée avec succès !');
      form.reset();
      document.getElementById('inputDate').valueAsDate = new Date();
      updateBadges();
    }

    // ---------- LOGIQUE ACTIVES ----------
    function loadActives() {
      activeDataCache = serverMock.getActiveCallSheets();
      document.getElementById('badgeActives').innerText = activeDataCache.length;
      renderActiveFilters();
      renderActivesTable(activeDataCache);
    }

    function renderActiveFilters() {
      const stats = {};
      
      activeDataCache.forEach(item => {
        const zap = String(item.ZAP || item.UAP || '').trim().toUpperCase();
        if (!zap) return;

        if (!stats[zap]) {
          stats[zap] = { total: 0, pair: 0, impair: 0 };
        }

        stats[zap].total++;
        const msnStr = String(item.MSN || '').replace(/[^0-9]/g, '');
        if (msnStr) {
          const msnNum = parseInt(msnStr, 10);
          if (msnNum % 2 === 0) stats[zap].pair++;
          else stats[zap].impair++;
        }
      });

      const container = document.getElementById('activeFiltersContainer');
      let html = `<div class="f-card f-card-all selected" onclick="applyFilter('ALL', 'ALL', this, event)">
                    <div style="font-size:12px;">TOUTES</div>
                    <div style="font-size:24px;">${activeDataCache.length}</div>
                  </div>`;

      Object.keys(stats).sort().forEach(zap => {
        html += `
          <div class="f-card">
            <div class="f-main" onclick="applyFilter('${zap}', 'ALL', this, event)">${zap} (${stats[zap].total})</div>
            <div class="f-sub">
              <div class="f-pill" onclick="applyFilter('${zap}', 'PAIR', this, event)">Pair: ${stats[zap].pair}</div>
              <div class="f-pill" onclick="applyFilter('${zap}', 'IMPAIR', this, event)">Impair: ${stats[zap].impair}</div>
            </div>
          </div>`;
      });

      container.innerHTML = html;
    }

    function applyFilter(zap, parity, elem, event) {
      if(event) event.stopPropagation();
      document.querySelectorAll('.f-main, .f-pill, .f-card-all').forEach(el => el.classList.remove('selected'));
      elem.classList.add('selected');
      
      let filtered = activeDataCache;
      
      if (zap !== 'ALL') {
        filtered = filtered.filter(item => String(item.ZAP || item.UAP || '').trim().toUpperCase() === zap);
      }
      
      if (parity !== 'ALL') {
        filtered = filtered.filter(item => {

          return parity === 'PAIR' ? (msnNum % 2 === 0) : (msnNum % 2 !== 0);
        });
      }
      
      renderActivesTable(filtered);
    }

    function sortArchives(colKey, isInitialLoad = false) {
      if (!isInitialLoad) {
        if (sortConfig.key === colKey) sortConfig.asc = !sortConfig.asc;
        else { sortConfig.key = colKey; sortConfig.asc = true; }
      }

      document.querySelectorAll('#tableArchives th.sortable').forEach(th => {
        let baseText = th.innerText.replace(/ ↑| ↓/g, ''); 
        if (th.getAttribute('data-col') === sortConfig.key) {
          th.innerText = baseText + (sortConfig.asc ? ' ↑' : ' ↓');
        } else {
          th.innerText = baseText;
        }
      });

      archiveDataCache.sort((a, b) => {
        let valA = a[sortConfig.key] || '';
        let valB = b[sortConfig.key] || '';
        let mod = sortConfig.asc ? 1 : -1;

        if (sortConfig.key === 'ID') {
          let numA = parseInt(String(valA).replace(/\D/g, '')) || 0;
          let numB = parseInt(String(valB).replace(/\D/g, '')) || 0;
          return (numA - numB) * mod;
        }
        if (sortConfig.key === 'MSN' || sortConfig.key === 'OF') {
          let numA = parseInt(String(valA).replace(/\D/g, ''));
          let numB = parseInt(String(valB).replace(/\D/g, ''));
          if (!isNaN(numA) && !isNaN(numB) && numA !== numB) {
            return (numA - numB) * mod;
          }
        }
        return String(valA).localeCompare(String(valB)) * mod;
      });

      filterArchives();
    }

    function getISOWeek(date) {
        let target = new Date(date.valueOf());
        let dayNr = (date.getDay() + 6) % 7;
        target.setDate(target.getDate() - dayNr + 3);
        let firstThursday = target.valueOf();
        target.setMonth(0, 1);
        if (target.getDay() !== 4) target.setMonth(0, 1 + ((4 - target.getDay()) + 7) % 7);
        return 1 + Math.ceil((firstThursday - target) / 604800000);
    }

    function calculateKpis() {
      const activeData = serverMock.getActiveCallSheets();
      const archiveData = serverMock.getArchivedCallSheets();
      const allData = [...activeData, ...archiveData];

      let mcaCount = 0;
      let finitionCount = 0;
      let systemeCount = 0;

      activeData.forEach(item => {
        const zap = String(item.ZAP || item.UAP || '').toUpperCase();
        if (zap === 'MCA') mcaCount++;
        else if (zap === 'FINITION') finitionCount++;
        else if (zap === 'SYSTÈME') systemeCount++;
      });

      let weeklyStats = {};
      allData.forEach(item => {
        const dateStr = item.Date_Creation || item.Date_Cloture;
        if (!dateStr) return;
        
        const dateVal = new Date(dateStr);
        if (isNaN(dateVal)) return;

        const zap = String(item.ZAP || item.UAP || '').toUpperCase();
        if (['MCA', 'FINITION', 'SYSTÈME'].includes(zap)) {
          const week = `Semaine ${getISOWeek(dateVal)}`;
          if (!weeklyStats[week]) weeklyStats[week] = { 'MCA': 0, 'FINITION': 0, 'SYSTÈME': 0 };
          weeklyStats[week][zap]++;
        }
      });

      return {
        MCA: mcaCount,
        FINITION: finitionCount,
        SYSTÈME: systemeCount,
        TOTAL_ARCHIVES: archiveData.length,
        weeklyStats: weeklyStats
      };
    }

    function loadKpis() {
      const d = calculateKpis();
      
      document.getElementById('kpi-mca').innerText = d.MCA || 0;
      document.getElementById('kpi-finition').innerText = d.FINITION || 0;
      document.getElementById('kpi-systeme').innerText = d.SYSTÈME || 0;
      document.getElementById('kpi-archives').innerText = d.TOTAL_ARCHIVES || 0;

      const ctx = document.getElementById('weeklyChart').getContext('2d');
      const weeks = Object.keys(d.weeklyStats || {}).sort();
      const dataMCA = weeks.map(w => d.weeklyStats[w].MCA);
      const dataFinition = weeks.map(w => d.weeklyStats[w].FINITION);
      const dataSysteme = weeks.map(w => d.weeklyStats[w].SYSTÈME);

      const chartData = {
        labels: weeks,
        datasets: [
          { label: 'MCA', data: dataMCA, backgroundColor: '#00205B', stack: 'Stack 0' },
          { label: 'FINITION', data: dataFinition, backgroundColor: '#0055A5', stack: 'Stack 0' },
          { label: 'SYSTÈME', data: dataSysteme, backgroundColor: '#E6A23C', stack: 'Stack 0' }
        ]
      };

      if (weeklyChartInstance) {
        weeklyChartInstance.data = chartData;
        weeklyChartInstance.update();
      } else {
        weeklyChartInstance = new Chart(ctx, {
          type: 'bar',
          data: chartData,
          options: { responsive: true, maintainAspectRatio: false, scales: { x: { stacked: true }, y: { stacked: true, beginAtZero: true, ticks: { stepSize: 1 } } }, plugins: { legend: { position: 'top' }, tooltip: { mode: 'index', intersect: false } } }
        });
      }
    }
